function _faF(v){
  v=Number(v)||0;
  var neg=v<0, a=Math.abs(v), t;
  if(a>=1e6) t='R$ '+(a/1e6).toLocaleString('pt-BR',{minimumFractionDigits:2,maximumFractionDigits:2})+' mi';
  else       t='R$ '+a.toLocaleString('pt-BR',{maximumFractionDigits:0});
  return (neg?'−':'')+t;
}

// Janela do período escolhido → [inicio, fim] ou null para "tudo".
function _faJanela(){
  if(_FA.de || _FA.ate){
    var a=_FA.de?new Date(_FA.de+'T00:00:00'):new Date(1900,0,1);
    var b=_FA.ate?new Date(_FA.ate+'T00:00:00'):new Date(2999,11,31);
    return [a,b];
  }
  var h=new Date(); h.setHours(0,0,0,0);
  switch(_FA.preset){
    case 'mes':    return [new Date(h.getFullYear(),h.getMonth(),1), new Date(h.getFullYear(),h.getMonth()+1,0)];
    case 'mesant': return [new Date(h.getFullYear(),h.getMonth()-1,1), new Date(h.getFullYear(),h.getMonth(),0)];
    case 'ano':    return [new Date(h.getFullYear(),0,1), new Date(h.getFullYear(),11,31)];
    case 'anoant': return [new Date(h.getFullYear()-1,0,1), new Date(h.getFullYear()-1,11,31)];
    case '12m':    return [new Date(h.getFullYear(),h.getMonth()-11,1), new Date(h.getFullYear(),h.getMonth()+1,0)];
    default:       return null;
  }
}

// v40: três problemas resolvidos de uma vez, trocando o <input type=date>

var _FA_ULTIMO_DIA=[31,28,31,30,31,30,31,31,30,31,30,31];
function _faFimDoMes(ano,mes){          // mes 1-12
  if(mes===2 && ((ano%4===0 && ano%100!==0) || ano%400===0)) return 29;
  return _FA_ULTIMO_DIA[mes-1];
}
// 'dd/mm/aaaa' → 'aaaa-mm-dd' com o dia corrigido. Devolve '' se não der.
function _faParse(txt){
  var m=String(txt||'').trim().match(/^(\d{1,2})\D(\d{1,2})\D(\d{4})$/);
  if(!m) return '';
  var d=+m[1], mes=+m[2], ano=+m[3];
  if(mes<1||mes>12||d<1||ano<1900||ano>2999) return '';
  var lim=_faFimDoMes(ano,mes);
  if(d>lim) d=lim;                      // 31/11 → 30/11 · 30/02 → 28/02
  return ano+'-'+String(mes).padStart(2,'0')+'-'+String(d).padStart(2,'0');
}
function _faParaBR(iso){
  var m=String(iso||'').match(/^(\d{4})-(\d{2})-(\d{2})$/);
  return m? m[3]+'/'+m[2]+'/'+m[1] : '';
}

// devolve mm/aaaa quando o intervalo atual é exatamente um mês inteiro
function _faMesAtual(){
  if(!_FA.de||!_FA.ate) return '';
  var a=_FA.de.match(/^(\d{4})-(\d{2})-01$/); if(!a) return '';
  var fim=_faFimDoMes(+a[1],+a[2]);
  if(_FA.ate!==a[1]+'-'+a[2]+'-'+String(fim).padStart(2,'0')) return '';
  return a[2]+'/'+a[1];
}
// v44: o botão "Mês" saiu. O campo mm/aaaa entra no mesmo "Aplicar" das
// datas — devolve true quando conseguiu aplicar um mês inteiro.
function _faAplicarMes(){
  var el=$('faMesIn'), msg=$('faMsg');
  var t=el?el.value.trim():'';
  if(!t) return false;
  var m=t.match(/^(\d{1,2})\D(\d{4})$/);
  if(!m || +m[1]<1 || +m[1]>12){
    if(msg) msg.textContent='Informe o mês como mm/aaaa — ex.: 05/2026.';
    return 'erro';
  }
  var mes=+m[1], ano=+m[2];
  if(ano<1900||ano>2999){ if(msg) msg.textContent='Ano fora do intervalo.'; return 'erro'; }
  var mm=String(mes).padStart(2,'0');
  _FA.de = ano+'-'+mm+'-01';
  _FA.ate= ano+'-'+mm+'-'+String(_faFimDoMes(ano,mes)).padStart(2,'0');
  _FA.preset='';
  if(msg) msg.textContent='';
  if($('faDe'))$('faDe').value=''; if($('faAte'))$('faAte').value='';
  return true;
}

function faAplicarIntervalo(){
  var eDe=$('faDe'), eAte=$('faAte'), msg=$('faMsg');
  var tDe=eDe?eDe.value.trim():'', tAte=eAte?eAte.value.trim():'';
  var aviso=function(t){ if(msg) msg.textContent=t||''; };

  // mm/aaaa preenchido tem prioridade — é o atalho para o mês inteiro
  var r=_faAplicarMes();
  if(r==='erro') return;
  if(r===true){ _faAtualizar(); return; }

  if(!tDe && !tAte){                       // os dois vazios = tirar o intervalo
    if(!_FA.de && !_FA.ate){ aviso(''); return; }
    _FA.de=''; _FA.ate=''; _FA.preset='ano'; aviso(''); _faAtualizar(); return;
  }
  var de=_faParse(tDe), ate=_faParse(tAte);
  if(!de || !ate){ aviso('Preencha as duas datas, no formato dd/mm/aaaa.'); return; }
  if(de>ate){ var t=de; de=ate; ate=t; }   // invertidas: troca em vez de reclamar
  // devolve a data já corrigida para o campo, para você VER o ajuste
  eDe.value=_faParaBR(de); eAte.value=_faParaBR(ate);
  var corrigiu = (_faParaBR(de)!==tDe) || (_faParaBR(ate)!==tAte);
  aviso(corrigiu?'Data ajustada para o último dia do mês.':'');
  if(de===_FA.de && ate===_FA.ate) return;
  _FA.de=de; _FA.ate=ate; _FA.preset='';
  _faAtualizar();
}

function _faDetalhe(){
  var r=_faFiltrar(_acCliente()?['Receita','A Receber']:['Receita','A Receber','Prejuízo']);
  var k=_FA.ord, sinal=_FA.asc?1:-1;
  return r.slice().sort(function(a,b){
    if(k==='valor') return sinal*(_faVal(a)-_faVal(b));
    if(k==='quem')  return sinal*_faQuem(a).localeCompare(_faQuem(b));
    if(k==='grupo') return sinal*String(a.grupo||'').localeCompare(String(b.grupo||''));
    var da=_faData(a),db=_faData(b);
    return sinal*((da?da.getTime():0)-(db?db.getTime():0));
  });
}

// ── Situação: PAGO / OK / VENCIDA, com a cor de cada uma ──────────
function _faSitTxt(f){
  var t=String(f.situacao||'').trim();
  if(t) return t;
  if(f.aba==='Receita' || String(f.pagamento||'').toUpperCase()==='SIM') return 'Pago';
  var d=pDate(f.vencimento), h=new Date(); h.setHours(0,0,0,0);
  if(d && d<h) return 'Vencida';
  return 'OK';
}
function _faSitTag(f){
  var t=_faSitTxt(f), l=t.toLowerCase(), cls='tn';
  if(l.indexOf('pag')===0||l==='quitado'||l==='recebido'||l==='quitada') cls='tg';
  else if(l.indexOf('venc')>=0||l.indexOf('atras')>=0) cls='tr';
  else if(l==='ok'||l.indexOf('em dia')>=0) cls='tb2';
  else if(l.indexOf('guia')>=0||l.indexOf('aberto')>=0||l.indexOf('parcial')>=0) cls='ta';
  else if(l.indexOf('prejuí')>=0||l.indexOf('prejui')>=0||l.indexOf('baixa')>=0) cls='tx';
  return '<span class="tag '+cls+'">'+esc(t.toUpperCase())+'</span>';
}

function _faTabelaPessoas(pessoas){
  var cliente=_acCliente();
  var linhas=pessoas.map(function(p){
    var salvo=_FA.quem; _FA.quem=[p];
    var rec=_faFiltrar(['Receita']), ar=_faFiltrar(['A Receber']), pj=_faFiltrar(['Prejuízo']);
    _FA.quem=salvo;
    return {p:p, rec:_faSoma(rec), n:rec.length, ar:_faSoma(ar), pj:_faSoma(pj),
            tk:rec.length?_faSoma(rec)/rec.length:0};
  });
  var tot=linhas.reduce(function(s,l){return s+l.rec;},0)||1;
  linhas.sort(function(a,b){return b.rec-a.rec;});
  var nCols=cliente?6:7;
  return '<div class="tw"><table><thead><tr>'
   +'<th>Pessoa</th><th style="text-align:right">Recebido</th><th style="text-align:right">%</th>'
   +'<th style="text-align:right">Lanç.</th><th style="text-align:right">Ticket médio</th>'
   +'<th style="text-align:right">A receber</th>'
   +(cliente?'':'<th style="text-align:right">Prejuízo</th>')
   +'</tr></thead><tbody>'
   + (linhas.length? linhas.map(function(l){
       var c=_faCor(l.p);
       return '<tr style="background:'+c.fundo+'55"><td>'+_faSelo(l.p)+'</td>'
        +'<td class="mono" style="text-align:right;color:var(--green-d);font-weight:600">'+_faFT(l.rec)+'</td>'
        +'<td class="mono" style="text-align:right">'+(l.rec/tot*100).toFixed(1)+'%</td>'
        +'<td class="mono" style="text-align:right">'+l.n+'</td>'
        +'<td class="mono" style="text-align:right">'+_faFT(l.tk)+'</td>'
        +'<td class="mono" style="text-align:right">'+(l.ar?_faFT(l.ar):'—')+'</td>'
        +(cliente?'':'<td class="mono" style="text-align:right'+(l.pj?';color:var(--red-d)':'')+'">'+(l.pj?_faFT(l.pj):'—')+'</td>')
        +'</tr>';
     }).join('') : '<tr><td colspan="'+nCols+'" class="vazio">Nada no período.</td></tr>')
   + '</tbody></table></div>';
}

// ── Cor por pessoa ───────────────────────────────────────────────
// v41: as cores vieram da planilha — verde para o Pedro, rosa para a
// Emanuelle. Elas funcionam muito bem como FUNDO (é para isso que
// servem no Sheets), mas como traço de gráfico sobre branco ficam quase
// invisíveis: #D4EDBC tem contraste ~1,2:1. Então cada pessoa tem duas
// cores da mesma família: a clara para preencher (selo, tinta da linha)
// e uma fechada para a barra do gráfico e o texto.
var _FA_PESSOA={
  'Pedro':     {fundo:'#D4EDBC', marca:'#4E9A2F', texto:'#2F6B1A'},
  'Emanuelle': {fundo:'#FFCFC9', marca:'#D2544A', texto:'#9B2C22'},
  'Escritório':{fundo:'#FBE9A8', marca:'#C9A84C', texto:'#8A6D14'},
  'Adriana':   {fundo:'#DBEAFE', marca:'#2563EB', texto:'#1D4ED8'}
};
var _FA_NEUTRAS=['#1B2A4A','#7C3AED','#0D9488','#DC2626','#0EA5E9','#6B7280'];
function _faCor(nome){
  var c=_FA_PESSOA[String(nome||'').trim()];
  if(c) return c;
  var h=0, t=String(nome||'');
  for(var i=0;i<t.length;i++) h=(h*31+t.charCodeAt(i))|0;
  var m=_FA_NEUTRAS[Math.abs(h)%_FA_NEUTRAS.length];
  return {fundo:m+'22', marca:m, texto:m};
}
// v45: no gráfico empilhado, as quatro cores "fechadas" das pessoas
// (verde-mato, tijolo, dourado e azul-forte) brigavam entre si. Agora a
// barra usa um tom intermediário entre o pastel da planilha e a cor
// fechada — sálvia, rosé, ouro-velho e azul-lavanda —, com um filete da
// cor fechada para separar as faixas. Mesma identidade, sem estridência.
function _faMistura(a,b,t){
  var h=function(x){ return [parseInt(x.slice(1,3),16),parseInt(x.slice(3,5),16),parseInt(x.slice(5,7),16)]; };
  var A=h(a), B=h(b), o='#';
  for(var i=0;i<3;i++) o+=('0'+Math.round(A[i]+(B[i]-A[i])*t).toString(16)).slice(-2);
  return o;
}
function _faCorBarra(nome){
  var c=_faCor(nome);
  if(!c._barra) c._barra=_faMistura(c.fundo,c.marca,.45);
  return c._barra;
}
function _faSelo(nome){
  var c=_faCor(nome);
  return '<span class="fa-pessoa" style="background:'+c.fundo+';color:'+c.texto+'">'+esc(nome)+'</span>';
}

function _faGraficos(receita){
  if(typeof Chart==='undefined')return;
  var pessoas=[...new Set(receita.map(_faQuem))].sort();
  // v44: sem animação. Cada repintura reanimava as barras do zero e dava a
  // impressão de que a tela estava recarregando sozinha.
  var SEM_ANIM={duration:0};

  // ── recebido mês a mês, empilhado por pessoa ──
  var meses=[...new Set(receita.map(function(f){var d=_faData(f);return d?_faMesKey(d):null;}).filter(Boolean))].sort();
  if($('cFaMes')){
    var ds=pessoas.map(function(p){
      return { label:p, backgroundColor:_faCorBarra(p), borderColor:'#ffffff',
        borderWidth:1.5, borderRadius:5, borderSkipped:false,
        hoverBackgroundColor:_faCor(p).marca,
        data:meses.map(function(m){
          return receita.filter(function(f){var d=_faData(f);return d&&_faMesKey(d)===m&&_faQuem(f)===p;})
                        .reduce(function(s,f){return s+_faVal(f);},0);
        })};
    });
    var totMes=meses.map(function(m){
      return receita.filter(function(f){var d=_faData(f);return d&&_faMesKey(d)===m;})
                    .reduce(function(s,f){return s+_faVal(f);},0);
    });
    mCh('cFaMes',{type:'bar',data:{labels:meses.map(_faMesLbl),datasets:ds},
      options:{responsive:true,maintainAspectRatio:false,animation:SEM_ANIM,
        interaction:{mode:'index',intersect:false},
        plugins:{legend:{display:pessoas.length>1,position:'bottom',
          labels:{color:'#4B5563',font:{size:10.5},boxWidth:11,usePointStyle:true,pointStyle:'circle',padding:10}},
          tooltip:{mode:'index',intersect:false,
            callbacks:{
              label:function(c){ return c.raw? ' '+c.dataset.label+': '+fF(c.raw) : null; },
              footer:function(items){
                if(!items||!items.length) return '';
                return 'Total do mês: '+fF(totMes[items[0].dataIndex]||0);
              }}}},
        scales:{x:{stacked:true,ticks:{color:'#9CA3AF',font:{size:10}},grid:{display:false}},

                   grid:{color:'rgba(0,0,0,.04)'}}}}});
  }

  // ── por tipo de serviço ──
  if($('cFaTipo')){
    var tp=[...new Set(receita.map(function(f){return f.tipo||'—';}))];
    var tv=tp.map(function(t){return receita.filter(function(f){return (f.tipo||'—')===t;}).reduce(function(s,f){return s+_faVal(f);},0);});
    var it=tv.map(function(_,i){return i;}).sort(function(a,b){return tv[b]-tv[a];}).slice(0,10);
    var cfgT=bCfg(it.map(function(i){return tp[i];}),it.map(function(i){return tv[i];}),null);
    cfgT.options.animation=SEM_ANIM;
    mCh('cFaTipo',cfgT);
  }

  // ── maiores grupos ──
  if($('cFaGrupo')){
    var gp=[...new Set(receita.map(function(f){return f.grupo||'—';}))];
    var gv=gp.map(function(g){return receita.filter(function(f){return (f.grupo||'—')===g;}).reduce(function(s,f){return s+_faVal(f);},0);});
    var ig=gv.map(function(_,i){return i;}).sort(function(a,b){return gv[b]-gv[a];}).slice(0,10);
    var cfgG=bCfg(ig.map(function(i){return gp[i];}),ig.map(function(i){return gv[i];}),null,true);
    cfgG.options.animation=SEM_ANIM;
    mCh('cFaGrupo',cfgG);
  }
}

function _popMesAnoSel(selId, rows, dateField, curVal) {
  var sel = document.getElementById(selId);
  if (!sel) return;
  var lb = {'01':'Jan','02':'Fev','03':'Mar','04':'Abr','05':'Mai','06':'Jun',
            '07':'Jul','08':'Ago','09':'Set','10':'Out','11':'Nov','12':'Dez'};
  var meses = [];
  var seen = {};
  rows.forEach(function(f){
    var d = pDate(f[dateField] || f.vencimento || f.dataPagamento);
    if (!d) return;
    var key = d.getFullYear() + '-' + String(d.getMonth()+1).padStart(2,'0');
    if (!seen[key]) { seen[key] = true; meses.push(key); }
  });
  meses.sort().reverse();
  sel.innerHTML = '<option value="">— Mês/Ano —</option>' +
    meses.map(function(m){
      var p = m.split('-');
      return '<option value="' + m + '"' + (m === curVal ? ' selected' : '') + '>' +
             (lb[p[1]] || p[1]) + '/' + p[0] + '</option>';
    }).join('');
  if (curVal) sel.value = curVal;
}
