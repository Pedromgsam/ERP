function _mapHeaders(headerRow) {
  var norm = function(s) {
    return String(s||'').trim().toLowerCase()
      .normalize('NFD').replace(/[̀-ͯ]/g,'');
  };
  var headers = headerRow.map(norm);
  var col = function(names) {
    var arr = Array.isArray(names) ? names : [names];
    for (var n = 0; n < arr.length; n++) {
      var idx = headers.indexOf(norm(arr[n]));
      if (idx >= 0) return idx;
    }
    return -1;
  };
  return { col: col, headers: headers };
}

// Normaliza texto para comparação de grupo/empresa: sem acento, sem caixa,
// sem espaço duplicado. É o que impede "Jorge Filho" e "jorge  filho" de
// virarem dois grupos distintos na hora de filtrar.
function _norm(s) {
  return String(s||'').trim().toLowerCase()
    .normalize('NFD').replace(/[̀-ͯ]/g,'').replace(/\s+/g,' ');
}
function _soDigitos(s) { return String(s||'').replace(/\D/g,''); }

function doGet(e) {
  try {
    var p = (e && e.parameter) ? e.parameter : {};
    if (p._auth === '1' && p.payload) {
      var pl = JSON.parse(Utilities.newBlob(Utilities.base64Decode(decodeURIComponent(p.payload))).getDataAsString());
      return handleAuth(pl);
    }
    if (p._wb === '1' && p.payload) {
      var pl2 = JSON.parse(Utilities.newBlob(Utilities.base64Decode(decodeURIComponent(p.payload))).getDataAsString());
      return handleWB(pl2);
    }
    return handleRead(p);
  } catch(err) { return jsonOk({ ok:false, erro:err.message }); }
}

function doPost(e) {
  try { return handleWB(JSON.parse(e.postData.contents)); }
  catch(err) { return jsonOk({ ok:false, erro:err.message }); }
}

// SESSÃO — resolve quem está pedindo e o que pode ver
// ════════════════════════════════════════════════════════════
//
// Devolve { nivel:'admin'|'cliente'|'negado', grupos:[...], login:'' }.
// A fonte é a aba "Portal do Cliente"; coluna Grupos = 'admin' dá acesso
// total, qualquer outro valor é lista de grupos separada por vírgula.
// Usuário inativo ou sem grupos → negado (falha fechada).
function _resolverSessao(login) {
  login = str(login || '').trim();

  if (!login) {
    // Sem identificação: em modo estrito, nega. Fora dele, mantém o
    // comportamento da v25 para não quebrar o ERP antes do ajuste no HTML.
    return MODO_ESTRITO
      ? { nivel:'negado', grupos:[], login:'', motivo:'Login não informado.' }
      : { nivel:'admin',  grupos:[], login:'', legado:true };
  }

  // v28: sem cache aqui também. A permissão é lida da planilha a cada
  // requisição — mudar o grupo de um usuário passa a valer no ato, e não
  // depois de 5 minutos. A aba "Portal do Cliente" é pequena; o custo é
  // desprezível perto de garantir que ninguém fique vendo o que já não pode.
  var sessao = { nivel:'negado', grupos:[], login:login, motivo:'Usuário não encontrado.' };
  try {
    var aba = _getPortalSheet();
    if (aba) {
      var d = aba.getDataRange().getValues();
      for (var i = 1; i < d.length; i++) {
        if (_norm(d[i][_PCOL.login]) !== _norm(login)) continue;

        var gruposRaw = str(d[i][_PCOL.grupos]);
        var ativo     = str(d[i][_PCOL.ativo]).toUpperCase();

        if (ativo !== 'SIM') {
          sessao = { nivel:'negado', grupos:[], login:login, motivo:'Usuário inativo.' };
          break;
        }
        if (gruposRaw.toLowerCase() === 'admin') {
          sessao = { nivel:'admin', grupos:[], login:login };
          break;
        }
        // Acesso "Contabilidade" -- ver MODULOS_STAFF_CONTAB, acima no arquivo.
        // Gatilho é Grupos Autorizados (não Login) -- tem que ficar sempre
        // igual à checagem espelhada no front (isStaffContab, em
        // _acLoginUnificado), senão front e backend discordam sobre o nível.
        if (_norm(gruposRaw) === 'contabilidade') {
          sessao = { nivel:'staffContab', grupos:[], login:login };
          break;
        }
        var lista = gruposRaw.split(',').map(function(g){ return g.trim(); })
                             .filter(function(g){ return !!g; });
        sessao = lista.length
          ? { nivel:'cliente', grupos:lista, login:login }
          // Cadastro ativo mas sem grupo definido: não é admin e não tem
          // carteira. Nega, em vez de devolver tudo por omissão.
          : { nivel:'negado', grupos:[], login:login, motivo:'Usuário sem grupo atribuído.' };
        break;
      }
    }
  } catch(e) {
    sessao = { nivel:'negado', grupos:[], login:login, motivo:'Erro ao resolver sessão.' };
  }

  return sessao;
}

function _agora() { return new Date().getTime(); }

function _usandoLoteDeLeitura() {
  try { return (typeof Sheets !== 'undefined' && !!Sheets.Spreadsheets); }
  catch(e) { return false; }
}

// Linhas vindas da API chegam "irregulares" (células vazias no fim da linha
// são omitidas). O código dos leitores indexa por posição, então iguala todas
// à largura da maior linha — do contrário um parcelamento perderia colunas.
function _emparelhar(linhas) {
  if (!linhas || !linhas.length) return [];
  var larg = 0, i;
  for (i = 0; i < linhas.length; i++) {
    if (linhas[i] && linhas[i].length > larg) larg = linhas[i].length;
  }
  for (i = 0; i < linhas.length; i++) {
    var l = linhas[i] || [];
    while (l.length < larg) l.push('');
    linhas[i] = l;
  }
  return linhas;
}

// Lê VÁRIAS abas de uma planilha numa única chamada.
// Devolve { 'NomeDaAba': [[linha],[linha]...] }.
// abas === null → descobre os nomes e lê todas.
function _lerAbasEmLote(spreadsheetId, abas) {
  var out = {};

  if (!_usandoLoteDeLeitura()) {
    // Modo antigo, usado só se a Sheets API não estiver ativada.
    var ss = SpreadsheetApp.openById(spreadsheetId);
    var lista = abas || ss.getSheets().map(function(sh){ return sh.getName(); });
    lista.forEach(function(nome) {
      var sh = ss.getSheetByName(nome);
      if (sh) out[nome] = _emparelhar(sh.getDataRange().getValues());
    });
    return out;
  }

  if (!abas) {
    // 1 chamada só para descobrir os nomes das abas (campo mínimo)
    var meta = Sheets.Spreadsheets.get(spreadsheetId, { fields: 'sheets.properties.title' });
    abas = (meta.sheets || []).map(function(sh){ return sh.properties.title; });
  }
  if (!abas.length) return out;

  // Aspas simples internas precisam ser dobradas na notação A1
  var ranges = abas.map(function(n){ return "'" + String(n).replace(/'/g, "''") + "'"; });

  var resp = Sheets.Spreadsheets.Values.batchGet(spreadsheetId, {
    ranges: ranges,
    valueRenderOption: 'FORMATTED_VALUE',        // "R$ 8.000,00" e "15/08/2026",
    dateTimeRenderOption: 'FORMATTED_STRING'     // exatamente como toNum/fmtDate esperam
  });

  (resp.valueRanges || []).forEach(function(vr, idx) {
    var nome = abas[idx];
    // O range devolvido vem com o nome real da aba; usa ele quando disponível
    if (vr.range) {
      var m = String(vr.range).match(/^'?(.*?)'?!/);
      if (m && m[1]) nome = m[1].replace(/''/g, "'");
    }
    out[nome] = _emparelhar(vr.values || []);
  });
  return out;
}

function handleRead(p) {
  if (p.acao === 'ping') return jsonOk({ ok:true, msg:'pong', versao:'v30', lote:_usandoLoteDeLeitura() });
  if (p.diag === '1')   return jsonOk(diagnosticarCarga());

  // Sistema Jurídico: acesso próprio, restrito à Base de Dados
  if (p.app === 'SJ') {
    if (str(p.chave) !== SJ_CHAVE) return jsonOk({ ok:false, erro:'Chave inválida.', _negado:true });
    var mSJ = str(p.modulo || 'baseDados');
    if (mSJ === 'clientes') mSJ = 'baseDados';
    if (SJ_MODULOS.indexOf(mSJ) < 0) return jsonOk({ ok:false, erro:'Módulo indisponível.' });
    return jsonOk({ ok:true, dados: safe(function(){ return _leitores()[mSJ](); }) });
  }

  var sessao = _resolverSessao(p.login);
  if (sessao.nivel === 'negado') {
    return jsonOk({ ok:false, erro:'Acesso negado. ' + (sessao.motivo||''), _negado:true });
  }
  var ehCliente = (sessao.nivel === 'cliente');
  var ehStaffContab = (sessao.nivel === 'staffContab');
  var t0Total = _agora();

  // Quais módulos devolver.
  //  · ?modulos=a,b,c  → carga sob demanda (o ERP.html usa para carregar em
  //    duas etapas e mostrar a tela antes de ter tudo)
  //  · ?modulo=x       → um só (compatibilidade)
  //  · nada            → todos os permitidos
  var pedidos;
  if (p.modulos)     pedidos = String(p.modulos).split(',').map(function(x){return x.trim();}).filter(Boolean);
  else if (p.modulo && p.modulo !== 'todos') pedidos = [String(p.modulo).trim()];
  else               pedidos = _CACHE_MODULOS.slice();

  pedidos = pedidos.filter(function(k){ return _CACHE_MODULOS.indexOf(k) >= 0; });
  if (ehCliente) pedidos = pedidos.filter(function(k){ return MODULOS_CLIENTE.indexOf(k) >= 0; });
  if (ehStaffContab) pedidos = pedidos.filter(function(k){ return MODULOS_STAFF_CONTAB.indexOf(k) >= 0; });
  if (!pedidos.length) return jsonOk({ ok:false, erro:'Nenhum módulo válido solicitado.' });

  // baseDados entra sempre que houver parcelamentos a recortar: é a chave que
  // liga um parcelamento ao seu grupo. Entra como insumo, não como resposta.
  var precisaBD = ehCliente && pedidos.indexOf('parcelamentos') >= 0 && pedidos.indexOf('baseDados') < 0;
  var aLer = precisaBD ? pedidos.concat(['baseDados']) : pedidos;

  var LEITORES = _leitores();
  var completo = {}, tempos = {};
  aLer.forEach(function(k) {
    var t = _agora();
    completo[k] = safe(function(){ return LEITORES[k](); });
    tempos[k] = _agora() - t;
  });

  var dados = {};
  if (ehCliente) {
    var ctx = { baseDados: completo.baseDados || [] };
    pedidos.forEach(function(k) {
      dados[k] = _recortarModulo(k, completo[k], sessao.grupos, ctx);
    });
    dados._nivel  = 'cliente';
    dados._grupos = sessao.grupos;
  } else if (ehStaffContab) {
    // Acesso "Contabilidade" -- ver MODULOS_STAFF_CONTAB. Sem recorte por
    // grupo e sem campos ocultados: vê o módulo liberado (financeiroContab)
    // por inteiro.
    pedidos.forEach(function(k){ dados[k] = completo[k]; });
    dados._nivel = 'staffContab';
  } else {
    pedidos.forEach(function(k){ dados[k] = completo[k]; });
    dados._nivel = 'admin';
  }

  dados.geradoEm = new Date().toISOString();
  dados._diag = {
    msTotal: _agora() - t0Total,
    msPorModulo: tempos,
    modulos: pedidos,
    leituraEmLote: _usandoLoteDeLeitura()   // false = Sheets API não ativada
  };
  return jsonOk(dados);
}

// ════════════════════════════════════════════════════════════
// RECORTE POR CLIENTE
// ════════════════════════════════════════════════════════════
//
// Filtra as linhas pelo grupo e apaga os campos sensíveis.
// Falha fechada: registro cujo grupo não pôde ser determinado é descartado.
function _recortarModulo(nome, linhas, grupos, ctx) {
  if (!linhas || !Array.isArray(linhas)) return [];
  if (!grupos || !grupos.length) return [];

  var permitidos = {};
  grupos.forEach(function(g){ permitidos[_norm(g)] = true; });

  var filtradas;
  if (nome === 'parcelamentos') {
    var idx = _indexarGruposPorEmpresa((ctx && ctx.baseDados) || []);
    filtradas = linhas.filter(function(p) {
      var g = _grupoDoParcelamento(p, idx, permitidos);
      return !!g && permitidos[_norm(g)] === true;
    });
  } else {
    var campo = CAMPO_GRUPO[nome] || 'grupo';
    filtradas = linhas.filter(function(r) {
      return r && permitidos[_norm(r[campo])] === true;
    });
  }

  var ocultos = CAMPOS_OCULTOS_CLIENTE[nome] || [];
  if (!ocultos.length) return filtradas;

  return filtradas.map(function(r) {
    var copia = {};
    for (var k in r) {
      if (!r.hasOwnProperty(k)) continue;
      if (ocultos.indexOf(k) >= 0) continue;   // removido do JSON, não escondido
      copia[k] = r[k];
    }
    return copia;
  });
}

// Índice de nome normalizado e CNPJ (só dígitos) → grupo, montado a partir
// da Base de Dados. É o que permite ligar um parcelamento ao seu grupo sem
// depender de o nome da aba coincidir com o nome do grupo.
function _indexarGruposPorEmpresa(baseDados) {
  var porNome = {}, porDoc = {};
  (baseDados || []).forEach(function(b) {
    if (!b || !b.grupo) return;
    var n = _norm(b.nome);
    if (n) porNome[n] = b.grupo;
    var doc = _soDigitos(b.cpfCnpj);
    if (doc.length >= 11) porDoc[doc] = b.grupo;
  });
  return { porNome:porNome, porDoc:porDoc };
}

function _grupoDoParcelamento(p, idx, permitidos) {
  if (!p) return '';
  var doc = _soDigitos(p.cnpj);
  if (doc && idx.porDoc[doc]) return idx.porDoc[doc];          // 1º: CNPJ (confiável)
  var n = _norm(p.empresa);
  if (n && idx.porNome[n]) return idx.porNome[n];              // 2º: nome exato
  // 3º: nome da aba, aceito só se bater exatamente com um grupo do usuário.
  // Sem esse último critério, parcelamento de empresa não cadastrada na Base
  // de Dados sumiria da tela do próprio dono.
  var aba = _norm(p.aba);
  if (aba && permitidos[aba] === true) return p.aba;
  return '';
}

function authAutenticar(payload) {
  var login = str(payload.email || payload.login || '').trim();
  if (!login) return { ok:false, erro:'Login não informado.' };
  var aba = _getPortalSheet();
  if (!aba) return { ok:false, erro:'Aba "Portal do Cliente" não encontrada.' };
  var data = aba.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    // v60: comparacao trocada de toLowerCase() cru pra _norm() -- a mesma
    // funcao ja usada em _resolverSessao() pro mesmo campo Login. Motivo:
    // toLowerCase() cru exige bater ate a forma Unicode exata do acento
    // (e/E + acento combinado x 'E' pre-composto podem parecer identicos na
    // tela e ainda assim serem strings diferentes); _norm() tira acento,
    // espaco duplicado e caixa, entao 'Eder'/'eder'/'ÉDER' etc. sempre
    // batem com o que esta na planilha, sem essa armadilha de encoding.
    if (_norm(data[i][_PCOL.login]) !== _norm(login)) continue;
    var grupos = str(data[i][_PCOL.grupos]).trim();
    var ativo  = str(data[i][_PCOL.ativo]).trim().toUpperCase();
    var isAdmin = grupos.toLowerCase() === 'admin';
    var isPend  = !isAdmin && ativo === 'SIM' && !grupos;
    try {
      aba.getRange(i+1, _PCOL.ultimoAcesso+1).setValue(
        Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm:ss'));
    } catch(e2) {}
    if (isPend) {
      var rawP = str(data[i][_PCOL.senhaHash]);
      var hashP = (rawP.length===64 && /^[0-9a-f]+$/.test(rawP)) ? rawP : _sha256(rawP);
      return { ok:true, usuario:{ login:login, senhaHash:hashP, grupos:'', ativo:ativo, status:'pendente' } };
    }
    var rawSenha = str(data[i][_PCOL.senhaHash]);
    var senhaHash = (rawSenha.length===64 && /^[0-9a-f]+$/.test(rawSenha)) ? rawSenha : _sha256(rawSenha);
    return { ok:true, usuario:{ login:login, senhaHash:senhaHash, grupos:grupos,
      ativo:ativo, nome:login, isAdmin:isAdmin, status:'ativo' }};
  }
  return { ok:false, usuario:null };
}

function _sha256(text) {
  var bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, text, Utilities.Charset.UTF_8);
  return bytes.map(function(b){ return (b<0?b+256:b).toString(16).padStart(2,'0'); }).join('');
}

      for (i2 = 0; i2 < parcelas.length; i2++) {
        var pa = parcelas[i2];
        var dv = _dataBR(pa.vencimento);
        var paga = String(pa.pagamento||'').toUpperCase() === 'SIM';
        if (!paga && dv && dv < hoje) vencidas++;
        if (!paga && dv && dv >= hoje && (!proxVenc || dv < _dataBR(proxVenc))) proxVenc = pa.vencimento;
        if (!dv || (dv >= limIni && dv <= limFim)) {
          // 'emissao' não é lida por tela nenhuma e 'aba' repete a do bloco
          naJanela.push({ numero:pa.numero, vencimento:pa.vencimento,
                          pagamento:pa.pagamento, status:pa.status });
        }
      }

      res.push({
        aba:nomeAba, empresa:empresa, cnpj:cnpj, local:local,
        natureza:natureza, numero:numero,
        totalParcelas:totalParc, parcelasPagas:parcPagas,
        valorUltimaParcela:valUltima, residual:residual,
        // contagens sobre o cronograma COMPLETO
        pagasReais: pagas,
        totalReais: parcelas.length,
        proximoVencimento: proxVenc,
        vencidas: vencidas,
        janela: true,
        parcelas: naJanela
      });
    }
  }
  return res;
}

function criarRascunhoEmail(d) {
  d = d || {};
  try {
    GmailApp.createDraft(str(d.to || ''), str(d.assunto || '(sem assunto)'), str(d.corpo || ''));

}

// Envia por e-mail as credenciais de acesso ao Portal. Usado pela tela de
// usuários. v29: só e-mail — o canal WhatsApp dependia de uma API que nunca
// esteve configurada (CFG.wpp_api_url era string vazia), então toda tentativa
// por lá falhava silenciosamente.
function enviarCredenciais(d) {
  var login = str(d && d.login || '');
  if (!login) return { ok:false, erro:'Login obrigatório.' };

  var email = '';
  var aba = _getPortalSheet();
  if (aba) {
    var data = aba.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      if (_norm(data[i][_PCOL.login]) === _norm(login)) { email = str(data[i][_PCOL.email]); break; }
    }
  }
  if (!email) return { ok:false, erro:'Sem e-mail cadastrado para ' + login + '.' };

  var msg = 'Olá! Suas credenciais de acesso ao Portal Araújo & Castro:\n'
          + 'Login: ' + login + '\n'
          + (d.senhaTemporaria ? 'Senha temporária: ' + str(d.senhaTemporaria) + '\n' : '')
          + 'Acesse: ' + str(d.urlPortal || '') + '\n'
          + 'Araújo & Castro Advocacia.';
  try {
    GmailApp.sendEmail(email, '[A&C] Acesso ao Portal — Credenciais', msg);
  } catch (e) {
    return { ok:false, erro:'Erro ao enviar: ' + e.message };
  }
  return { ok:true, msg:'Credenciais enviadas para ' + email + '.' };
}

  return { ok:true, versao:'v30', leituraEmLote:_usandoLoteDeLeitura(),
           msTotal:_agora()-t0, modulos:mods };
}
